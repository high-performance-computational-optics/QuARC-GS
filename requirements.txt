#   pip install torch==2.11.0 torchvision==0.26.0 --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements.txt --no-build-isolation
#
# The three git dependencies (libgs, diff-gaussian-rasterization, simple-knn) are NOT listed here --
# they are pinned as git submodules under third_party/ and installed from there. See README section 1b.
